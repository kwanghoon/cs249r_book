More things to install!

1. Quarto 

wget https://github.com/quarto-dev/quarto-cli/releases/download/v1.8.26/quarto-1.8.26-linux-amd64.deb
sudo dpkg -i quarto-1.8.26-linux-amd64.deb 

2. R and some packages

sudo apt update -qq
sudo apt install --no-install-recommends software-properties-common dirmngr
wget -qO- https://cloud.r-project.org/bin/linux/ubuntu/marutter_pubkey.asc | sudo tee -a /etc/apt/trusted.gpg.d/cran_ubuntu_key.asc
sudo add-apt-repository "deb https://cloud.r-project.org/bin/linux/ubuntu $(lsb_release -cs)-cran40/"
sudo apt install --no-install-recommends r-base

sudo Rscript -e "install.packages(c('knitr', 'rmarkdown'), repos='https://cran.rstudio.com/')"

sudo Rscript tools/dependencies/install_packages.R
